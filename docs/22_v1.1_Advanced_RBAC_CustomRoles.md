# Ellines EIP v1.1 — Advanced RBAC (Custom Roles & Attribute-Based Access)

**Status:** 📋 DESIGN DOCUMENT  
**Date:** August 1, 2026  
**Scope:** Custom role builder, attribute-based access control, fine-grained permissions

---

## Overview

**Current State:** 6 fixed roles (owner, admin, executive, manager, member, viewer).

**v1.1 Goal:** Let Owner create **custom roles** with:
- ✅ Custom role templates (Finance Manager, Operations Lead, Analyst, etc.)
- ✅ Fine-grained permissions (can read, create, approve, delete)
- ✅ Attribute-based access (department, region, cost center)
- ✅ Resource-level permissions (can access Glance but not Admin)
- ✅ Time-based permissions (can approve only 9am–5pm)
- ✅ Delegation (can temporarily grant permissions to another user)

**Key Principle:** RBAC goes from **"broad roles"** → **"capability-based access"**.

## Architecture

### Permission Model (Fine-Grained)

```typescript
// Current (v1.0): Role-based
type UserRole = 'owner' | 'admin' | 'executive' | 'manager' | 'member' | 'viewer';

// v1.1: Capability-based
type Permission = 
  // Org Management
  | 'org:read' | 'org:write' | 'org:rename' | 'org:suspend' | 'org:invite_user' | 'org:remove_user'
  
  // Connectors
  | 'connector:read' | 'connector:install' | 'connector:test' | 'connector:sync' | 'connector:delete'
  
  // Workflows
  | 'approval:read' | 'approval:create' | 'approval:decide' | 'approval:escalate'
  | 'rule:read' | 'rule:create' | 'rule:execute' | 'rule:delete'
  | 'report:read' | 'report:create' | 'report:run' | 'report:delete'
  
  // Dashboards
  | 'dashboard:read' | 'dashboard:create' | 'dashboard:publish' | 'dashboard:delete'
  
  // Data Access
  | 'data:read:people' | 'data:read:fleet' | 'data:read:documents'
  | 'data:write:documents' | 'data:delete:documents'
  
  // System
  | 'audit:read' | 'settings:read' | 'settings:write' | 'admin:access';

type ResourcePermission = {
  permission: Permission;
  resources?: string[];  // Specific resource IDs (e.g., ['report-1', 'report-2'])
  attributes?: {          // Attribute-based (department, region, etc.)
    department?: string;
    region?: string;
    costCenter?: string;
  };
  conditions?: {          // Time/context-based
    timeRange?: { start: string; end: string };  // '09:00' – '17:00'
    requiresMfa?: boolean;
    ipWhitelist?: string[];
  };
};
```

```prisma
// Enhanced User/Membership to support custom roles
model OrganizationMembership {
  id             String   @id @default(cuid())
  userId         String   @map("user_id")

  role           UserRole @default(member)  // Still here for backward compat

  elevatedRole   String?  @map("elevated_role")   // Temporary higher role
  elevatedUntil  DateTime? @map("elevated_until")  // Expiration
  elevatedBy     String?  @map("elevated_by")     // Who granted it
  
  // Delegation
  delegatedPermissions Json? @map("delegated_permissions")  // Grant specific permissions to another user
  delegatedUntil DateTime? @map("delegated_until")
  
  // Status & Audit
  isActive       Boolean  @default(true) @map("is_active")
  joinedAt       DateTime @default(now()) @map("joined_at")
  invitedBy      String?  @map("invited_by")
  
  createdAt      DateTime @default(now()) @map("created_at")
  updatedAt      DateTime @updatedAt @map("updated_at")

  user           User         @relation(fields: [userId], references: [id], onDelete: Cascade)
  organization   Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
  customRole     CustomRole?  @relation(fields: [customRoleId], references: [id], onDelete: SetNull)

  @@unique([userId, organizationId])
  @@index([userId])
  @@index([organizationId])
  @@index([customRoleId])
  @@map("organization_memberships")
}

  id             String   @id @default(cuid())
```

## Permission Evaluation Engine

```typescript
// services/identity/src/rbac/permission-evaluator.service.ts

export class PermissionEvaluatorService {
  /**
   * Check if user has permission to perform action.
   * Considers: role, custom role, elevation, delegation, attributes, conditions.
   */
  async hasPermission(
    userId: string,
    orgId: string,
    permission: Permission,
    context?: {
      resourceId?: string;
      attributes?: Record<string, string>;
      ip?: string;
      timestamp?: Date;
    },
  ): Promise<{ allowed: boolean; reason?: string }> {
    // Step 1: Get user's membership
    const membership = await prisma.organizationMembership.findUnique({
      where: { userId_organizationId: { userId, organizationId: orgId } },
      include: { customRole: true },
    });

    if (!membership || !membership.isActive) {
      return { allowed: false, reason: 'User is not a member of this org' };
    }

    // Step 2: Check if elevated
    if (membership.elevatedUntil && membership.elevatedUntil > new Date()) {
      // User is temporarily elevated
      const elevatedPermissions = await this.getPermissionsForRole(
        membership.elevatedRole || membership.role,
        orgId,
        membership.customRoleId,
      );

      if (this.hasPermissionInSet(permission, elevatedPermissions, context)) {
        return { allowed: true };
      }
    }

    // Step 3: Check delegated permissions
    if (membership.delegatedPermissions && membership.delegatedUntil > new Date()) {
      const delegated = membership.delegatedPermissions as DelegatedPermission[];
      if (delegated.some(p => this.matchesPermission(p, permission, context))) {
        return { allowed: true };
      }
    }

    // Step 4: Check role-based permissions
    const rolePermissions = await this.getPermissionsForRole(
      membership.role,
      orgId,
      membership.customRoleId,
    );

    if (!this.hasPermissionInSet(permission, rolePermissions, context)) {
      return { allowed: false, reason: 'Permission denied by role' };
    }

    // Step 5: Evaluate conditions (time, IP, MFA, attributes)
    const condition = this.getPermissionCondition(permission, rolePermissions);
    
    if (condition?.timeRange) {
      const now = context?.timestamp || new Date();
      const [start, end] = condition.timeRange;
      if (!this.isWithinTimeRange(now, start, end)) {
        return { allowed: false, reason: 'Permission denied: outside allowed time' };
      }
    }

    if (condition?.requiresMfa && !context?.mfaVerified) {
      return { allowed: false, reason: 'MFA required' };
    }

    if (condition?.ipWhitelist && context?.ip) {
      if (!this.isIpInWhitelist(context.ip, condition.ipWhitelist)) {
        return { allowed: false, reason: 'IP not whitelisted' };
      }
    }

    // Step 6: Check attribute-based access
    if (condition?.attributes) {
      if (!this.attributesMatch(context?.attributes, condition.attributes)) {
        return { allowed: false, reason: 'Attributes do not match' };
      }
    }

    return { allowed: true };
  }

  /**
   * Get all permissions for a role (custom or built-in).
   */
  private async getPermissionsForRole(
    role: string,
    orgId: string,
    customRoleId?: string | null,
  ): Promise<ResourcePermission[]> {
    if (customRoleId) {
      const customRole = await prisma.customRole.findUnique({
        where: { id: customRoleId },
      });
      if (customRole) {
        return (customRole.permissions as ResourcePermission[]) || [];
      }
    }

    // Fall back to built-in role
    return this.getBuiltInRolePermissions(role);
  }

  private getBuiltInRolePermissions(role: string): ResourcePermission[] {
    const builtInRoles: Record<string, ResourcePermission[]> = {
      owner: [
        // Full access
        { permission: 'org:read' },
        { permission: 'org:write' },
        { permission: 'org:suspend' },
        { permission: 'org:invite_user' },
        { permission: 'connector:read' },
        { permission: 'connector:install' },
        { permission: 'approval:read' },
        { permission: 'approval:decide' },
        { permission: 'rule:read' },
        { permission: 'rule:create' },
        { permission: 'admin:access' },
        // ... all permissions
      ],
      admin: [
        { permission: 'org:read' },
        { permission: 'org:write' },
        { permission: 'org:invite_user' },
        { permission: 'connector:read' },
        { permission: 'connector:install' },
        { permission: 'approval:read' },
        { permission: 'approval:decide' },
        { permission: 'rule:read' },
        { permission: 'admin:access' },
        // ... most permissions except org:suspend
      ],
      manager: [
        { permission: 'org:read' },
        { permission: 'approval:read' },
        { permission: 'approval:decide' },
        { permission: 'approval:escalate' },
        { permission: 'report:read' },
        { permission: 'dashboard:read' },
        { permission: 'data:read:people' },
        // ... middle-level permissions
      ],
      member: [
        { permission: 'org:read' },
        { permission: 'approval:read' },
        { permission: 'report:read' },
        { permission: 'dashboard:read' },
        { permission: 'data:read:people' },
        { permission: 'data:read:fleet' },
        // ... basic read access
      ],
      viewer: [
        { permission: 'org:read' },
        { permission: 'dashboard:read' },
        { permission: 'report:read' },
        // ... read-only access
      ],
    };

    return builtInRoles[role] || [];
  }

  private hasPermissionInSet(
    permission: Permission,
    permissionSet: ResourcePermission[],
    context?: { resourceId?: string; attributes?: Record<string, string> },
  ): boolean {
    return permissionSet.some(p => {
      if (p.permission !== permission) return false;
      if (p.resources && context?.resourceId) {
        return p.resources.includes(context.resourceId);
      }
      if (p.attributes && context?.attributes) {
        return Object.entries(p.attributes).every(
          ([key, value]) => context.attributes?.[key] === value,
        );
      }
      return true;  // No resource/attribute restrictions
    });
  }
}
```

```typescript
function PermissionPicker({ permissions, onChange }) {
  const allPermissions: Permission[] = [
    'org:read', 'org:write', 'org:invite_user',
    'connector:read', 'connector:install', 'connector:sync',
    'approval:read', 'approval:decide', 'approval:escalate',
    'rule:read', 'rule:create', 'rule:execute',
    'report:read', 'report:create', 'report:run',
    'dashboard:read', 'dashboard:create', 'dashboard:publish',
    'data:read:people', 'data:read:fleet',
    'audit:read', 'settings:read', 'admin:access',
  ];

  return (
    <div className="permission-grid">
      {allPermissions.map(perm => (
        <label key={perm}>
          <input
            type="checkbox"
            checked={permissions.some(p => p.permission === perm)}
            onChange={(e) => {
              const updated = e.target.checked
                ? [...permissions, { permission: perm }]
                : permissions.filter(p => p.permission !== perm);
              onChange(updated);
            }}
          />
          {perm}
        </label>
      ))}
    </div>
  );
}
```

## Migration Path (v1.0 → v1.1)

```typescript
// Migrate existing users to custom roles (backward compatible)
async function migrateUsersToCustomRoles() {
  const users = await prisma.user.findMany();

  for (const user of users) {
    const membership = await prisma.organizationMembership.findUnique({
      where: { userId_organizationId: { userId: user.id, organizationId: user.organizationId } },
    });

    if (membership) {
      // Keep membership.role as-is (built-in)
      // Set membership.customRoleId to null
      // On login, evaluate permissions using hasPermission()
      
      await prisma.organizationMembership.update({
        where: { id: membership.id },
        data: { customRoleId: null },
      });
    }
  }
}
```

## Security Best Practices

✅ **Always check permissions on the backend** (not just frontend)  
✅ **Log all role changes** (audit trail)  
✅ **Require MFA for sensitive permissions** (like org:suspend)  
✅ **Expire elevated roles automatically**  
✅ **Limit delegation duration** (max 7 days)  
✅ **Prevent privilege escalation** (users can't upgrade themselves)  
✅ **Owner-only gate** (only Owner can manage roles)
