# Ellines EIP v1.1 — Autonomous Workflows (AI Agent Rules)

**Status:** 📋 DESIGN DOCUMENT  
**Date:** August 1, 2026  
**Scope:** AI-driven approvals and rules that execute without human intervention

---

## Overview

**Current State:** Rules + Approvals are manual (Owner/IT must decide).

**v1.1 Goal:** Let EIP automatically **execute workflows** based on:
- ✅ Business rule conditions (if healthScore < 50, auto-trigger alert)
- ✅ Ellinea AI recommendations (if Ellinea confidence > 90%, auto-approve)
- ✅ Scheduled actions (daily reconciliation, weekly report, monthly billing)
- ✅ Event-driven automation (when connector sync fails, email IT)
- ✅ Human guardrails (review required if decision is risky)

**Key Principle:** Autonomous = **not "rogue AI"** but **well-scoped, auditable, human-approved automation** that reduces busywork.

---

## Architecture

### Three Levels of Autonomy

#### Level 1: Pure Rules (Deterministic)
```javascript
if (healthScore < 50) {
  sendAlert("System health critical");
  escalateToOwner();
}
```
✅ No AI involved. Fully deterministic. Safe to run without approval.

#### Level 2: Conditional Ellinea (AI-Assisted)
```javascript
if (openAlerts > 5) {
  let recommendation = await ellinea.recommendAction();
  if (recommendation.confidence > 0.95) {
    execute(recommendation);  // Auto-approve
    auditLog({ action: 'auto_approve', confidence });
  } else {
    createApproval(recommendation);  // Human review
  }
}
```
✅ AI assists. Human can still review if low confidence. Auditable.

#### Level 3: Scheduled Workflows (Delegated)
```javascript
// Every night at 2am:
await dailyReconciliation();
await sendExecutiveBrief();
// Owner pre-approves this workflow template
```
✅ Scheduled tasks. Pre-approved template. Runs unattended.

```prisma
  createdBy      String   @map("created_by")
  approvedBy     String?  @map("approved_by")
  approvedAt     DateTime? @map("approved_at")
  executionLog   Json     @default("[]")  // Last 10 executions
  
  // State
  lastExecutedAt DateTime? @map("last_executed_at")
  failureCount   Int      @default(0) @map("failure_count")
  
  createdAt      DateTime @default(now()) @map("created_at")
  updatedAt      DateTime @updatedAt @map("updated_at")

  organization   Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)

  @@index([organizationId])
  @@index([schedule])  // For cron job lookup
  @@index([enabled])
  @@map("business_rules")
```

## Execution Engine

### Rule Evaluator (Runs Every 5 Minutes)

```typescript
// services/identity/src/workflows/rule-evaluator.service.ts

export class RuleEvaluatorService {
  /**
   * Evaluate all active rules for an org.
   * Called via scheduled job every 5 min.
   */
  async evaluateOrgRules(orgId: string) {
    const rules = await prisma.businessRule.findMany({
      where: { organizationId: orgId, enabled: true },
    });

    for (const rule of rules) {
      const triggered = await this.evaluateRuleTrigger(rule);
      if (triggered) {
        await this.executeRule(rule);
      }
    }
  }

  /**
   * Evaluate if rule trigger condition is met.
   */
  async evaluateRuleTrigger(rule: BusinessRule): Promise<boolean> {
    if (rule.when === 'healthScore_lt') {
      const snapshot = await this.getCurrentSnapshot(rule.organizationId);
      return snapshot.healthScore < (rule.threshold || 50);
    }

    if (rule.when === 'openAlerts_gt') {
      const snapshot = await this.getCurrentSnapshot(rule.organizationId);
      return snapshot.openAlerts > (rule.threshold || 10);
    }

    if (rule.when === 'schedule') {
      // Cron job already triggered via scheduler
      return true;
    }

    if (rule.when === 'event_type_matches') {
      // Check if recent events match pattern
      const pattern = new RegExp(rule.eventPattern || '');
      const recentEvents = await this.getRecentEvents(rule.organizationId);
      return recentEvents.some(e => pattern.test(e.type));
    }

    return false;
  }

  /**
   * Execute the rule action (deterministic or AI-assisted).
   */
  async executeRule(rule: BusinessRule) {
    const execution = await prisma.workflowExecution.create({
      data: {
        ruleId: rule.id,
        organizationId: rule.organizationId,
        status: 'pending',
        triggeredBy: 'schedule',
        triggeredAt: new Date(),
      },
    });

    try {
      // Level 1: Pure deterministic action
      if (rule.autonomyLevel === 'level1') {
        await this.executeAction(rule.action, rule.actionConfig);
        
        await prisma.workflowExecution.update({
          where: { id: execution.id },
          data: {
            status: 'success',
            completedAt: new Date(),
            result: { executed: true },
          },
        });
      }

      // Level 2: AI-assisted with human gate
      if (rule.autonomyLevel === 'level2') {
        const aiRec = await this.getEllineaRecommendation(rule);
        
        await prisma.workflowExecution.update({
          where: { id: execution.id },
          data: { aiRecommendation: aiRec },
        });

        if (
          aiRec.confidence >= (rule.ellinea?.minConfidence || 0.9) &&
          !rule.humanGate
        ) {
          // Auto-approve
          await this.executeAction(rule.action, rule.actionConfig);
          await prisma.workflowExecution.update({
            where: { id: execution.id },
            data: {
              status: 'success',
              aiApproved: true,
              aiApprovedAt: new Date(),
              completedAt: new Date(),
            },
          });
        } else if (rule.humanGate) {
          // Wait for human review
          await prisma.workflowExecution.update({
            where: { id: execution.id },
            data: { status: 'blocked_by_human_gate' },
          });

          // Notify Owner/IT
          await this.notifyForReview(execution, aiRec);
        }
      }

      // Level 3: Scheduled workflow
      if (rule.autonomyLevel === 'level3') {
        const result = await this.executeWorkflowSteps(rule);
        
        if (rule.humanGate && !rule.approvedBy) {
          // Block execution if not pre-approved
          await prisma.workflowExecution.update({
            where: { id: execution.id },
            data: { status: 'blocked_by_human_gate' },
          });
        } else {
          await prisma.workflowExecution.update({
            where: { id: execution.id },
            data: {
              status: 'success',
              completedAt: new Date(),
              result,
            },
          });
        }
      }
    } catch (err) {
      await prisma.workflowExecution.update({
        where: { id: execution.id },
        data: {
          status: 'failed',
          errorMessage: err instanceof Error ? err.message : 'Unknown error',
          completedAt: new Date(),
        },
      });

      // Alert Owner/IT on failure
      await this.notifyExecutionFailure(execution, err);
    }

    // Update rule's last execution time
    await prisma.businessRule.update({
      where: { id: rule.id },
      data: { lastExecutedAt: new Date() },
    });
  }

  /**
   * Get Ellinea AI recommendation for this situation.
   */
  async getEllineaRecommendation(rule: BusinessRule) {
    const context = {
      rule: rule.name,
      trigger: rule.when,
      org: rule.organizationId,
    };

    const response = await fetch('http://ellinea:3002/api/recommend-action', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ context }),
    });

    return await response.json() as {
      action: string;
      confidence: number;
      reasoning: string;
    };
  }

  /**
   * Execute the actual action (send alert, create approval, etc.).
   */
  async executeAction(action: string, config: unknown) {
    const cfg = config as Record<string, unknown>;
    
    if (action === 'send_alert') {
      const recipients = (cfg.recipients as string[]) || [];
      // Send email alert
      for (const email of recipients) {
        await this.mailService.send({
          to: email,
          subject: 'EIP Alert',
          body: cfg.message as string,
        });
      }
    }

    if (action === 'create_approval') {
      // Create approval request
      await prisma.approvalRequest.create({
        data: {
          organizationId: cfg.organizationId as string,
          title: cfg.title as string,
          detail: cfg.detail as string,
          requester: 'system@ellinea.eip',
          status: 'pending',
        },
      });
    }

    if (action === 'sync_connector') {
      // Trigger connector sync
      const connectorId = cfg.connectorId as string;
      await this.connectorService.syncConnector(connectorId);
    }

    if (action === 'run_report') {
      // Generate and email report
      const reportId = cfg.reportId as string;
      const report = await this.generateReport(reportId);
      await this.mailService.send({
        to: cfg.recipients as string[],
        subject: 'Report',
        body: report,
      });
    }
  }
}
```

### Cron Job Scheduler

```typescript
// services/identity/src/workflows/cron-scheduler.service.ts

import * as cron from 'node-cron';

export class CronSchedulerService {
  private jobs: Map<string, cron.ScheduledTask> = new Map();

  /**
   * Start scheduled workflows for all orgs.
   */
  async startScheduler(orgId: string) {
    const workflows = await prisma.scheduledWorkflow.findMany({
      where: { organizationId: orgId, isActive: true },
    });

    for (const wf of workflows) {
      const key = `${orgId}:${wf.id}`;

      // Unschedule if already running
      if (this.jobs.has(key)) {
        this.jobs.get(key)?.stop();
      }

      // Schedule new job
      const task = cron.schedule(wf.schedule, async () => {
        console.log(`[Cron] Running workflow ${wf.name}`);
        await this.executeScheduledWorkflow(wf);
      }, { timezone: wf.timezone });

      this.jobs.set(key, task);
    }
  }

  async executeScheduledWorkflow(workflow: ScheduledWorkflow) {
    const execution = await prisma.workflowExecution.create({
      data: {
        ruleId: workflow.id,
        organizationId: workflow.organizationId,
        status: 'running',
        triggeredBy: 'schedule',
        triggeredAt: new Date(),
      },
    });

    try {
      const result = await this.executeSteps(workflow.steps as WorkflowStep[]);

      await prisma.workflowExecution.update({
        where: { id: execution.id },
        data: {
          status: 'success',
          completedAt: new Date(),
          result,
        },
      });
    } catch (err) {
      await prisma.workflowExecution.update({
        where: { id: execution.id },
        data: {
          status: 'failed',
          errorMessage: err instanceof Error ? err.message : 'Unknown error',
          completedAt: new Date(),
        },
      });
    }

    await prisma.scheduledWorkflow.update({
      where: { id: workflow.id },
      data: { lastExecutedAt: new Date() },
    });
  }

  private async executeSteps(steps: WorkflowStep[]): Promise<unknown> {
    const results: unknown[] = [];

    for (const step of steps) {
      if (step.type === 'sync_connector') {
        await this.syncConnector(step.connectorId as string);
      }
      if (step.type === 'generate_report') {
        const report = await this.generateReport(step.reportId as string);
        results.push(report);
      }
      if (step.type === 'send_email') {
        await this.sendEmail(step.to as string[], step.subject as string);
        results.push({ emailSent: true });
      }
    }

    return { stepsExecuted: steps.length, results };
  }
}
```

## Guardrails & Safety

### Rule Safety Checklist

```typescript
// Validation before rule creation
function validateRule(rule: BusinessRule): string[] {
  const errors: string[] = [];

  // Level 1 rules: no guardrails needed (deterministic)
  if (rule.autonomyLevel === 'level1') {
    return errors;
  }

  // Level 2 rules: must have reasonable thresholds
  if (rule.autonomyLevel === 'level2') {
    if (!rule.ellinea?.minConfidence || rule.ellinea.minConfidence < 0.8) {
      errors.push('AI confidence threshold must be >= 80%');
    }
    if (rule.action === 'auto_approve' && !rule.humanGate) {
      errors.push('Human gate MUST be enabled for autonomous approvals');
    }
  }

  // Level 3 rules: must be pre-approved by Owner
  if (rule.autonomyLevel === 'level3') {
    if (!rule.approvedBy) {
      errors.push('Owner approval REQUIRED before scheduled workflow creation');
    }
  }

  return errors;
}
```
